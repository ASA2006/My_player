export default{async fetch(q){
const u=new URL(q.url).searchParams.get('url');
const c={'access-control-allow-origin':'*','access-control-allow-headers':'*'};
if(q.method==='OPTIONS')return new Response(null,{headers:c});
if(!u)return new Response('missing url',{status:400});
const h=new Headers(),g=q.headers.get('range');
if(g)h.set('range',g);
const r=await fetch(u,{headers:h,redirect:'follow'});
const o=new Headers(r.headers);
o.set('access-control-allow-origin','*');
o.set('access-control-expose-headers','*');
return new Response(r.body,{status:r.status,headers:o})}}
// ثم ضع: https://اسمك.workers.dev/?url={url}
ألصق رابطًا ثم اضغط «تشغيل». يدعم HLS وDASH وMPEG-TS وFLV وMP4 وقوائم M3U.
